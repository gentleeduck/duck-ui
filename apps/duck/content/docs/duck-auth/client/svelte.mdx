```ts title="src/lib/auth.ts"
import { createAuthStore } from '@gentleduck/auth/client/svelte'

export const auth = createAuthStore({ baseUrl: '/auth' })
```

It takes the [vanilla client's options](/duck-auth/client/vanilla), or a built one as `client`, and reads the session straight away; `noInitialFetch` skips that read and starts as a guest. `state` is a readable store of `{ session, identity, status }`, and needs no Svelte import to build.

```svelte title="src/routes/+page.svelte"
<script lang="ts">
  import { auth } from '$lib/auth'

  const state = auth.state
  let email = ''
  let password = ''
  let error: string | null = null

  async function submit() {
    const res = await auth.signIn({ providerId: 'password', input: { email, password } })
    error = res.ok ? null : res.error.code
  }
</script>

{#if $state.status === 'authed'}
  {$state.identity?.profile.email}
  <button on:click={() => auth.signOut()}>Sign out</button>
{:else if $state.status === 'guest'}
  <form on:submit|preventDefault={submit}>
    <input bind:value={email} type="email" required />
    <input bind:value={password} type="password" required />
    <button>Sign in</button>
    {#if error === 'AUTH_INVALID_CREDENTIALS'}<p>Wrong email or password</p>{/if}
  </form>
{/if}
```

`signIn`, `signOut` and `refresh` answer the call's [envelope](/duck-auth/client#what-a-call-answers). `auth.client` is the client itself, for `beginProvider` and `signUp`.
