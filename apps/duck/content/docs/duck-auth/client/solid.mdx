```tsx
import { Provider } from '@gentleduck/auth/client/solid'

export function App() {
  return (
    <Provider baseUrl="/auth">
      <Account />
    </Provider>
  )
}
```

`Provider` takes the [vanilla client's options](/duck-auth/client/vanilla), or a built one as `client`. It reads the session on mount; `noInitialFetch` skips that read and starts as a guest.

```tsx
import { authUseSession, authUseSignIn, authUseSignOut } from '@gentleduck/auth/client/solid'
import { createSignal, Match, Switch } from 'solid-js'

function Account() {
  const { data, status } = authUseSession()
  const signIn = authUseSignIn()
  const signOut = authUseSignOut()
  const [error, setError] = createSignal<string | null>(null)

  async function submit(event: SubmitEvent & { currentTarget: HTMLFormElement }) {
    event.preventDefault()
    const form = new FormData(event.currentTarget)
    const res = await signIn.mutate({
      providerId: 'password',
      input: { email: form.get('email'), password: form.get('password') },
    })
    setError(res.ok ? null : res.error.code)
  }

  return (
    <Switch>
      <Match when={status() === 'authed'}>
        {data().identity?.profile.email}
        <button type="button" onClick={() => signOut.mutate()}>Sign out</button>
      </Match>
      <Match when={status() === 'guest'}>
        <form onSubmit={submit}>
          <input name="email" type="email" required />
          <input name="password" type="password" required />
          <button type="submit" disabled={signIn.loading()}>Sign in</button>
          {error() === 'AUTH_INVALID_CREDENTIALS' && <p>Wrong email or password</p>}
        </form>
      </Match>
    </Switch>
  )
}
```

- `authUseSession()` answers `data` and `status` as signals, and `refresh()`.
- `authUseSignIn()` and `authUseSignOut()` answer `{ mutate, loading, error }`, the last two as signals. `mutate` resolves to the call's [envelope](/duck-auth/client#what-a-call-answers); `error` holds only a thrown error.
- `authUseClient()` is the client itself: `beginProvider` and `signUp` are called on it.
