```ts title="src/main.ts"
import { createAuthVuePlugin } from '@gentleduck/auth/client/vue'
import { createApp } from 'vue'
import App from './App.vue'

createApp(App).use(createAuthVuePlugin({ baseUrl: '/auth' })).mount('#app')
```

The plugin takes the [vanilla client's options](/duck-auth/client/vanilla), or a built one as `client`. It reads the session on install; `noInitialFetch` skips that read and starts as a guest.

```vue title="src/components/SignIn.vue"
<script setup lang="ts">
import { useAuthSession, useAuthSignIn, useAuthSignOut } from '@gentleduck/auth/client/vue'
import { ref } from 'vue'

const { data, status } = useAuthSession()
const signIn = useAuthSignIn()
const signOut = useAuthSignOut()
const email = ref('')
const password = ref('')
const error = ref<string | null>(null)

async function submit() {
  const res = await signIn.mutate({ providerId: 'password', input: { email: email.value, password: password.value } })
  error.value = res.ok ? null : res.error.code
}
</script>

<template>
  <p v-if="status === 'loading'">Loading</p>
  <div v-else-if="status === 'authed'">
    {{ data.identity?.profile.email }}
    <button @click="signOut.mutate()">Sign out</button>
  </div>
  <form v-else @submit.prevent="submit">
    <input v-model="email" type="email" required />
    <input v-model="password" type="password" required />
    <button :disabled="signIn.loading.value">Sign in</button>
    <p v-if="error === 'AUTH_INVALID_CREDENTIALS'">Wrong email or password</p>
  </form>
</template>
```

- `useAuthSession()` answers `data` and `status` as refs, and `refresh()`.
- `useAuthSignIn()` and `useAuthSignOut()` answer `{ mutate, loading, error }`. `mutate` resolves to the call's [envelope](/duck-auth/client#what-a-call-answers); `error` holds only a thrown error.
- `useAuthClient()` is the client itself: `beginProvider` and `signUp` are called on it.
