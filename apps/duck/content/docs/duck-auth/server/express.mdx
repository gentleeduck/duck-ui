```ts
import {
  expressActorContext,
  expressCsrf,
  mountProviderBegin,
  mountProviderCallback,
  mountSession,
  mountSignIn,
  mountSignOut,
} from '@gentleduck/auth/server/express'
import express from 'express'

const app = express()
app.set('trust proxy', 1)

app.post('/auth/signin', express.json(), mountSignIn(auth))
app.post('/auth/signout', mountSignOut(auth))
app.get('/auth/session', mountSession(auth))
app.post('/auth/providers/:id/begin', express.json(), mountProviderBegin(auth))
app.get('/auth/providers/:id/callback', mountProviderCallback(auth))
app.post('/auth/providers/:id/callback', express.urlencoded({ extended: false }), mountProviderCallback(auth))
```

- `express.json()` goes on the routes that read a body.
- `trust proxy` decides which IP `req.ip` answers, and so what the session records. Set it to the number of proxies in front of you.
- The callback answers the session cookies and an empty 200. To land the browser on a page, write the callback yourself, as on the [OAuth page](/duck-auth/providers/oauth#the-routes).

## Your own routes

```ts
import { toHeaders } from '@gentleduck/auth/server/express'

app.use('/api', expressCsrf(auth), expressActorContext(auth))

app.get('/api/me', async (req, res) => {
  const resolved = await auth.resolveSession({ headers: toHeaders(req.headers) }).orNull()
  if (!resolved) {
    res.status(401).end()
    return
  }
  res.json(resolved.identity)
})
```

- `expressCsrf` answers the 403 itself, so it does not depend on your error handler.
- `expressActorContext` passes a refusal to `next(err)`. Its `AuthError` carries `status` and `toJSON()` for your error handler.

## Calling a flow yourself

`applyIntents(intents, res)` writes a flow's answer, cookies included, onto the response. `expressCaller(req)` is the IP and User-Agent to stamp on the session:

```ts
import { applyIntents, expressCaller } from '@gentleduck/auth/server/express'

app.post('/auth/magic-link/redeem', express.json(), expressCsrf(auth), async (req, res, next) => {
  try {
    const { intents } = await auth.flows.signIn({
      providerId: 'magic-link',
      input: { token: String(req.body.token ?? '') },
      ...expressCaller(req),
    })
    applyIntents(intents, res)
  } catch (err) {
    next(err)
  }
})
```
